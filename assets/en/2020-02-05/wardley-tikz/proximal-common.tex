% Compact frames avoid the long default structural stage labels colliding in
% the multi-panel figures. The analytical guide positions remain unchanged.
\newcommand{\PVVerticalFrame}{%
  \draw[draw=black,line width=1.15pt,
    -{Latex[length=2.3mm,width=1.8mm]}] (0,0) -- (0,1);%
}

% A capped line makes the vertical projection explicit without suggesting
% that the Euclidean length of a diagonal dependency has a map meaning.
\newcommand{\PVGapGuide}[3]{%
  \draw[pv gap guide] (#1,#2) -- (#1,#3);%
  \draw[pv gap guide] ({#1-.012},#2) -- ({#1+.012},#2);%
  \draw[pv gap guide] ({#1-.012},#3) -- ({#1+.012},#3);%
}
